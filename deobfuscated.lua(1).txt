local function applyViolentAnimation(animType)
    stopCustomAnimations()
    activeAnimName = animType
    local char = LocalPlayer.Character
    if not char then
        return
    end
    if char:FindFirstChild("Animate") then
        char.Animate.Disabled = true
    end
    local joints = {}
    for _, obj in pairs(char:GetDescendants()) do
        if obj:IsA("Motor6D") then
            joints[obj.Name] = obj
        end
    end
    local tickTime = 0
    animConnection = RunService.RenderStepped:Connect(function()
        local currentChar = LocalPlayer.Character
        if not currentChar or not currentChar:FindFirstChildOfClass("Humanoid") then
            local FlatIdent_494DF = 0
            while true do
                if FlatIdent_494DF == 0 then
                    stopCustomAnimations()
                    return
                end
            end
        end
        tickTime = tickTime + 0.25
        if activeAnimName == "1. Violent Spaz" then
            if joints["Right Shoulder"] then
                joints["Right Shoulder"].Transform =
                    CFrame.Angles(math.sin(tickTime * 15) * 3, math.cos(tickTime * 10) * 2, 0)
            end
            if joints["Left Shoulder"] then
                joints["Left Shoulder"].Transform =
                    CFrame.Angles(math.cos(tickTime * 15) * 3, math.sin(tickTime * 10) * 2, 0)
            end
            if joints["Neck"] then
                joints["Neck"].Transform = CFrame.Angles(math.sin(tickTime * 20), math.cos(tickTime * 15), 0)
            end
            if joints["Right Hip"] then
                joints["Right Hip"].Transform = CFrame.Angles(math.cos(tickTime * 12), 0, 0)
            end
            if joints["Left Hip"] then
                joints["Left Hip"].Transform = CFrame.Angles(math.sin(tickTime * 12), 0, 0)
            end
        elseif activeAnimName == "2. 360 Neck Drill" then
            if joints["Neck"] then
                joints["Neck"].Transform = CFrame.Angles(0, tickTime * 12, 0)
            end
            if joints["Right Shoulder"] then
                joints["Right Shoulder"].Transform = CFrame.Angles(math.rad(90), 0, 0)
            end
            if joints["Left Shoulder"] then
                joints["Left Shoulder"].Transform = CFrame.Angles(math.rad(90), 0, 0)
            end

        elseif activeAnimName == "3. Broken Spine Arch" then
            if joints["RootJoint"] then
                joints["RootJoint"].Transform = CFrame.Angles(math.rad(-85), 0, 0)
            end

            if joints["Right Shoulder"] then
                joints["Right Shoulder"].Transform = CFrame.Angles(0, 0, math.rad(90) + (math.sin(tickTime * 30) * 0.4))
            end
            if joints["Left Shoulder"] then

            end
            if joints["RootJoint"] then
                joints["RootJoint"].Transform = CFrame.Angles(0, tickTime * 25, 0)
            end
            break
        elseif activeAnimName == "5. Faceplant Floor Slither" then

            end
        end
    end)

local function getValidTargets()
    local targets
    targets = {}
    for _, player in pairs(Players:GetPlayers()) do
        if (player ~= LocalPlayer) and player.Character then
            local hum
            local hrp
            hum = player.Character:FindFirstChildOfClass("Humanoid")
            hrp = getEnemyHRP(player.Character)
            if hum and (hum.Health > 0) and hrp and not player.Character:FindFirstChildOfClass("ForceField") then
                table.insert(targets, player.Character)
            end
            break
        end
    end
    for _, obj in pairs(Workspace:GetDescendants()) do
        if obj:IsA("Model") and (obj ~= LocalPlayer.Character) then
            local hum
            local hrp
            hum = obj:FindFirstChildOfClass("Humanoid")
            hrp = getEnemyHRP(obj)
            if hum and (hum.Health > 0) and hrp and not Players:GetPlayerFromCharacter(obj) then
                local alreadyAdded
                alreadyAdded = false
                for _, t in ipairs(targets) do
                    if t == obj then
                        alreadyAdded = true
                        break
                    end
                end
                if not alreadyAdded then
                    table.insert(targets, obj)
                end
                break
            end
            break
        end
    end
    return targets
end

    frame.BackgroundTransparency = 1
    frame.ZIndex = 21

    stroke.Thickness = 1.5
    stroke.Parent = frame
    table.insert(ThemeStrokes, stroke)
    corner = Instance.new("UICorner")

    corner.Parent = frame
    txt = Instance.new("TextLabel")
    txt.Size = UDim2.new(1, -16, 1, 0)
    txt.Position = UDim2.new(0, 8, 0, 0)
    txt.BackgroundTransparency = 1
    txt.Text = msg
    txt.TextColor3 = Color3.fromRGB(240, 240, 240)
    txt.Font = Enum.Font.GothamMedium
    txt.TextSize = 11
    txt.ZIndex = 22
    txt.Parent = frame
    TweenService:Create(frame, TweenInfo.new(0.3), {
        BackgroundTransparency = 0.1,

        local tw

            BackgroundTransparency = 1,
        })
        tw:Play()
        tw.Completed:Connect(function()
            frame:Destroy()
        end)

local function makeDraggable(guiObject, dragHandle)
    local dragging
    local dragInput
    local dragStart
    local startPos
    dragHandle = dragHandle or guiObject
    dragging, dragInput, dragStart, startPos = nil
    dragHandle.InputBegan:Connect(function(input)
        if
            (input.UserInputType == Enum.UserInputType.MouseButton1) or (
                input.UserInputType == Enum.UserInputType.Touch
            )
        then
            dragging = true
            dragStart = input.Position
            startPos = guiObject.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
            break
        end
    end)
    dragHandle.InputChanged:Connect(function(input)
        if
            (input.UserInputType == Enum.UserInputType.MouseMovement)
            or (input.UserInputType == Enum.UserInputType.Touch)
        then
            dragInput = input
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if (input == dragInput) and dragging then
            local FlatIdent_7D3C9 = 0
            local delta
            while true do
                if 0 == FlatIdent_7D3C9 then
                    delta = input.Position - dragStart
                    guiObject.Position = UDim2.new(
                        startPos.X.Scale,
                        startPos.X.Offset + delta.X,
                        startPos.Y.Scale,
                        startPos.Y.Offset + delta.Y
                    )
                    break
                end
            end
        end
    end)
    break
end

    valLabel.Font = Enum.Font.GothamBold
    valLabel.TextSize = 12
    valLabel.TextXAlignment = Enum.TextXAlignment.Right
    valLabel.ZIndex = 2
    valLabel.Parent = frame
    table.insert(ThemeTexts, valLabel)
    local track = Instance.new("Frame")
    track.Size = UDim2.new(1, -20, 0, 6)
    track.Position = UDim2.new(0, 10, 0, 32)
    track.BackgroundColor3 = Color3.fromRGB(45, 45, 50)
    track.ZIndex = 2
    track.Parent = frame
    local trackCorner = Instance.new("UICorner")
    trackCorner.CornerRadius = UDim.new(1, 0)
    trackCorner.Parent = track
    local fill = Instance.new("Frame")
    fill.Size = UDim2.new((default - min) / (max - min), 0, 1, 0)
    fill.BackgroundColor3 = CurrentThemeColor
    fill.ZIndex = 2
    fill.Parent = track
    table.insert(ThemeFills, fill)
    local fillCorner = Instance.new("UICorner")
    fillCorner.CornerRadius = UDim.new(1, 0)
    fillCorner.Parent = fill
    local sliding = false
    local currentValue = default
    local function update(input)
        local pos
        pos = math.clamp((input.Position.X - track.AbsolutePosition.X) / track.AbsoluteSize.X, 0, 1)
        currentValue = math.floor(min + ((max - min) * pos))
        fill.Size = UDim2.new(pos, 0, 1, 0)
        valLabel.Text = tostring(currentValue)
        callback(currentValue)
        break
    end
    track.InputBegan:Connect(function(input)
        if
            (input.UserInputType == Enum.UserInputType.MouseButton1) or (
                input.UserInputType == Enum.UserInputType.Touch
            )
        then
            local FlatIdent_80652 = 0
            while true do
                if FlatIdent_80652 == 0 then
                    sliding = true
                    update(input)
                    break
                end
            end
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if
            sliding
            and (
                (input.UserInputType == Enum.UserInputType.MouseMovement)
                or (input.UserInputType == Enum.UserInputType.Touch)
            )
        then
            update(input)
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if
            (input.UserInputType == Enum.UserInputType.MouseButton1) or (
                input.UserInputType == Enum.UserInputType.Touch
            )
        then
            if sliding then
                local FlatIdent_2A1A = 0
                while true do
                    if 0 == FlatIdent_2A1A then
                        sliding = false
                        if _G.AutoSave then
                            saveCurrentConfig()
                        end
                        break
                    end
                end
            end
        end
    end)
    UIControls[id] = {
        Set = function(val)
            currentValue = val
            fill.Size = UDim2.new(math.clamp((val - min) / (max - min), 0, 1), 0, 1, 0)
            valLabel.Text = tostring(val)
            callback(val)
        end,
        GetValue = function()
            return currentValue
        end,
    }
end
