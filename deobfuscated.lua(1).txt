--=====================================================================
-- helper
--=====================================================================
local function char()
    local c = LP.Character
    if not c then return end
    local h = c:FindFirstChildOfClass("Humanoid")
    local r = c:FindFirstChild("HumanoidRootPart")
    if not h or not r or h.Health <= 0 then return end
    return c, h, r
end

local function hrp(m)
    if not m then return nil end
    return m:FindFirstChild("HumanoidRootPart")
        or m:FindFirstChild("Torso")
        or m:FindFirstChild("UpperTorso")
end

local function fixCam()
    local c, h = char()
    if not h then return end
    h.CameraOffset = Vector3.zero
    local cam = workspace.CurrentCamera
    if cam then cam.CameraSubject = h end
end

--=====================================================================
-- NPC system
--=====================================================================
local function validNPC(n)
    if not n or not n.Parent then return false end
    if not n:IsDescendantOf(AllNoobs) then return false end
    return hrp(n) ~= nil
end

local function npcRoot(n)
    if not n or not n.Parent then return end
    local h = n:FindFirstChildOfClass("Humanoid")
    if not h or h.Health <= 0 then return end
    local r = hrp(n)
    if not r or not r:IsA("BasePart") then return end
    return r, h
end

local function pickNPC()
    if validNPC(S.npcTarget) then return S.npcTarget end
    S.npcTarget = nil
    for _, n in ipairs(AllNoobs:GetChildren()) do
        if n.Name == "HealerNoob" and validNPC(n) then
            S.npcTarget = n
            return n
        end
    end
    for _, n in ipairs(AllNoobs:GetChildren()) do
        if validNPC(n) then
            S.npcTarget = n
            return n
        end
    end
end

local function faceTP(root, pos, lookAt)
    local d = lookAt - pos
    if d.Magnitude > 0.001 then
        root.CFrame = CFrame.lookAt(pos, lookAt, Vector3.yAxis)
    else
        root.CFrame = CFrame.new(pos)
    end
end

local function npcTick(dt)
    if not S.npcOn then return end
    local c, h, r = char()
    if not r then return end
    fixCam()
    local n = pickNPC()
    if not n then
        r.CFrame = CFrame.new(0, 20, 0) * r.CFrame.Rotation
        return
    end
    local nr = npcRoot(n)
    if not nr then
        S.npcTarget = nil
        return
    end

    local target
    if S.posMode == "Top" then
        target = nr.Position + Vector3.new(0, S.dist, 0)
    elseif S.posMode == "Bottom" then
        target = nr.Position - Vector3.new(0, S.dist, 0)
    elseif S.posMode == "Orbit" then
        S.orbitAng += S.orbitSpd * dt
        target = nr.Position + Vector3.new(
            math.cos(S.orbitAng) * S.dist, 0,
            math.sin(S.orbitAng) * S.dist
        )
    end

    if target then faceTP(r, target, nr.Position) end
end

local function startNPC()
    if S.npcConn then S.npcConn:Disconnect() end
    S.npcConn = RunService.Heartbeat:Connect(npcTick)
end

local function stopNPC()
    if S.npcConn then
        S.npcConn:Disconnect()
        S.npcConn = nil
    end
    S.npcTarget = nil
end

--=====================================================================
-- hitbox
--=====================================================================
local function isTool(t)
    if not t or not t:IsA("Tool") then return false end
    return t:FindFirstChild("Handle") ~= nil
        or t:FindFirstChildOfClass("BasePart") ~= nil
end

local function updateHitbox(t)
    if not isTool(t) then return end
    local h = t:FindFirstChild("Handle") or t:FindFirstChildOfClass("BasePart")
    if not h then return end
    if S.orbit then return end

    if S.hitbox or S.afk then
        local sz = math.min(S.hitboxMul, 15)
        h.Size = Vector3.new(sz, sz, sz)
        h.Transparency = 0.8
        h.CanCollide = false
        h.Massless = true
    else
        h.Size = Vector3.new(2, 2, 2)
        h.Transparency = 0
    end
end

local function refreshHitbox()
    local c = LP.Character
    if not c then return end
    for _, t in ipairs(c:GetChildren()) do
        updateHitbox(t)
    end
end

local function bindChar(c)
    c.ChildAdded:Connect(function(x)
        if x:IsA("Tool") then updateHitbox(x) end
    end)
    for _, x in ipairs(c:GetChildren()) do
        if x:IsA("Tool") then updateHitbox(x) end
    end
end

if LP.Character then bindChar(LP.Character) end
LP.CharacterAdded:Connect(bindChar)

                                er.AssemblyAngularVelocity = Vector3.zero
                            end
                        end
                    end
                end
            end)
        end
    end
end)

local dashConn = nil
local function setupDash()
    if dashConn then dashConn:Disconnect() end
    dashConn = RunService.RenderStepped:Connect(function()
        if not S.dash then return end
        pcall(function()
            local g = LP:FindFirstChild("PlayerGui")
            local d = g and g:FindFirstChild("Dash")
            if d then
                local a = d:FindFirstChild("amount")
                if a and a:IsA("NumberValue") then a.Value = 1 end
            end
            if LP:GetAttribute("NoDashing") then
                LP:SetAttribute("NoDashing", false)
            end
        end)
    end)
end

            end)

local function getTargets()
    local list = {}
    local me = LP.Character
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LP and p.Character then
            local h = p.Character:FindFirstChildOfClass("Humanoid")
            local r = hrp(p.Character)
            if h and h.Health > 0 and r
                and not p.Character:FindFirstChildOfClass("ForceField") then
                table.insert(list, p.Character)
            end
        end
    end
    for _, o in ipairs(workspace:GetDescendants()) do
        if o:IsA("Model") and o ~= me then
            local h = o:FindFirstChildOfClass("Humanoid")
            local r = hrp(o)
            if h and h.Health > 0 and r
                and not Players:GetPlayerFromCharacter(o)
                and not o:FindFirstChildOfClass("ForceField") then
                local dup = false
                for _, t in ipairs(list) do
                    if t == o then dup = true break end
                end
                if not dup then table.insert(list, o) end
            end
        end
    end
    return list
end

task.spawn(function()
    while task.wait(0.05) do
        if S.afk then
            pcall(function()
                local c, h, r = char()
                if not r then return end

                if not S.afkFloat or S.afkFloat.Parent ~= r then
                    if S.afkFloat then S.afkFloat:Destroy() end
                    S.afkFloat = Instance.new("BodyVelocity")
                    S.afkFloat.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
                    S.afkFloat.Velocity = Vector3.zero
                    S.afkFloat.Parent = r
                end

                if r.Position.Y < 500 then
                    r.CFrame = r.CFrame + Vector3.new(0, 100, 0)
                end

                if not S.afkTarget
                    or not S.afkTarget.Parent
                    or not S.afkTarget:FindFirstChildOfClass("Humanoid")
                    or S.afkTarget:FindFirstChildOfClass("Humanoid").Health <= 0 then
                    if S.afkTarget then
                        local old = hrp(S.afkTarget)
                        if old then old.Anchored = false end
                    end
                    local list = getTargets()
                    if #list > 0 then
                        S.afkTarget = list[math.random(1, #list)]
                        local nr = hrp(S.afkTarget)
                        if nr then nr.Anchored = true end
                    else
                        S.afkTarget = nil
                    end
                end

                if S.afkTarget then
                    local tr = hrp(S.afkTarget)
                    local th = S.afkTarget:FindFirstChildOfClass("Humanoid")
                    if tr and th and th.Health > 0 then
                        tr.Anchored = true
                        tr.CFrame = r.CFrame * CFrame.new(0, 0, -2)
                    else
                        if tr then tr.Anchored = false end
                        S.afkTarget = nil
                    end
                end

                local tool = c:FindFirstChildOfClass("Tool")
                if not tool then
                    local bp = LP:FindFirstChild("Backpack")
                    if bp then
                        for _, x in ipairs(bp:GetChildren()) do
                            if x:IsA("Tool") then x.Parent = c tool = x break end
                        end
                    end
                end
                if tool then tool:Activate() end
            end)
        else
            if S.afkFloat then S.afkFloat:Destroy() S.afkFloat = nil end
            if S.afkTarget then
                local r = hrp(S.afkTarget)
                if r then r.Anchored = false end
                S.afkTarget = nil
            end
        end
    end
end)

local function stopAnim()
    if S.animConn then S.animConn:Disconnect() S.animConn = nil end
    S.animName = "None"
    local c = LP.Character
    if c then
        local a = c:FindFirstChild("Animate")
        if a then a.Disabled = false end
        for _, m in ipairs(c:GetDescendants()) do
            if m:IsA("Motor6D") then m.Transform = CFrame.identity end
        end
    end
end

local function playAnim(name)
    stopAnim()
    S.animName = name
    local c = LP.Character
    if not c then return end
    local a = c:FindFirstChild("Animate")
    if a then a.Disabled = true end

    local joints = {}
    for _, o in ipairs(c:GetDescendants()) do
        if o:IsA("Motor6D") then joints[o.Name] = o end
    end

    local t = 0
    S.animConn = RunService.RenderStepped:Connect(function()
        local cur = LP.Character
        if not cur or not cur:FindFirstChildOfClass("Humanoid") then
            stopAnim()
            return
        end
        t = t + 0.25
        local n = S.animName

        if n == "Spaz" then
            if joints["Right Shoulder"] then joints["Right Shoulder"].Transform = CFrame.Angles(math.sin(t*15)*3, math.cos(t*10)*2, 0) end
            if joints["Left Shoulder"] then joints["Left Shoulder"].Transform = CFrame.Angles(math.cos(t*15)*3, math.sin(t*10)*2, 0) end
            if joints["Neck"] then joints["Neck"].Transform = CFrame.Angles(math.sin(t*20), math.cos(t*15), 0) end
            if joints["Right Hip"] then joints["Right Hip"].Transform = CFrame.Angles(math.cos(t*12), 0, 0) end
            if joints["Left Hip"] then joints["Left Hip"].Transform = CFrame.Angles(math.sin(t*12), 0, 0) end

        elseif n == "Neck Drill" then
            if joints["Neck"] then joints["Neck"].Transform = CFrame.Angles(0, t*12, 0) end
            if joints["Right Shoulder"] then joints["Right Shoulder"].Transform = CFrame.Angles(math.rad(90), 0, 0) end
            if joints["Left Shoulder"] then joints["Left Shoulder"].Transform = CFrame.Angles(math.rad(90), 0, 0) end

        elseif n == "Broken Spine" then
            if joints["RootJoint"] then joints["RootJoint"].Transform = CFrame.Angles(math.rad(-85), 0, 0) end
            if joints["Neck"] then joints["Neck"].Transform = CFrame.Angles(math.rad(90), 0, 0) end
            if joints["Right Shoulder"] then joints["Right Shoulder"].Transform = CFrame.Angles(math.rad(180), 0, 0) end
            if joints["Left Shoulder"] then joints["Left Shoulder"].Transform = CFrame.Angles(math.rad(180), 0, 0) end

        elseif n == "Tpose" then
            if joints["Right Shoulder"] then joints["Right Shoulder"].Transform = CFrame.Angles(0, 0, math.rad(90) + math.sin(t*30)*0.4) end
            if joints["Left Shoulder"] then joints["Left Shoulder"].Transform = CFrame.Angles(0, 0, math.rad(-90) - math.sin(t*30)*0.4) end
            if joints["RootJoint"] then joints["RootJoint"].Transform = CFrame.Angles(0, t*25, 0) end

        elseif n == "Faceplant" then
            if joints["RootJoint"] then joints["RootJoint"].Transform = CFrame.Angles(math.rad(90), math.sin(t*10)*1.5, 0) end
            if joints["Neck"] then joints["Neck"].Transform = CFrame.Angles(math.rad(-45), 0, 0) end
            if joints["Right Shoulder"] then joints["Right Shoulder"].Transform = CFrame.Angles(math.rad(-90), 0, 0) end
            if joints["Left Shoulder"] then joints["Left Shoulder"].Transform = CFrame.Angles(math.rad(-90), 0, 0) end
        end
    end)
end

local function applyBigHead(on)
    local c = LP.Character
    if not c then return end
    local head = c:FindFirstChild("Head")
    if not head then return end
    local orig = head:FindFirstChild("OrigSize")
    if on then
        if not orig then
            local v = Instance.new("Vector3Value")
            v.Name = "OrigSize"
            v.Value = head.Size
            v.Parent = head
        end
        head.Size = head.Size * Vector3.new(2,2,2)
    else
        if orig then
            head.Size = orig.Value
            orig:Destroy()
        end
    end
end

    if not writefile then return end

        hitbox = S.hitbox, hitboxMul = S.hitboxMul,
        magnet = S.magnet, magnetDist = S.magnetDist,

        speed = S.speed, jumpPow = S.jumpPow,
        afk = S.afk, antiafk = S.antiafk,
        spin = S.spin, orbit = S.orbit,

    pcall(function()
        writefile(CFG, HttpService:JSONEncode(d))
    end)
end

    end
